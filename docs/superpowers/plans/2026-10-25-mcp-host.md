# MCP Host Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 给现有桌面 Agent 增加本地 stdio MCP 宿主：Python sidecar 启动 MCP Server、发现工具、按权限注册进 LangGraph，并在桌面端提供 MCP 状态面板；第一版预置 Playwright，断网或 Server 失败不影响核心功能。

**Architecture:** 新增 `mcp_client.py`（JSON-RPC over stdio）、`mcp_manager.py`（配置/生命周期/状态）、`mcp_tools.py`（MCP 工具转 LangChain Tool）。`graph_agent.build_graph` 接受 `extra_tools`；`desktop_api.create_app` 在 startup 后台启动 MCP、构建图时注入健康工具；React 右侧面板新增 MCP 标签页。MCP 完全可选，核心 `TOOLS` 不依赖它。

**Tech Stack:** Python 3.12、pytest/unittest、LangChain `StructuredTool`、FastAPI、React 18、TypeScript、Vite、Vitest、Electron、`@playwright/mcp`。

**Spec:** `docs/superpowers/specs/2026-10-25-mcp-host-design.md`

from __future__ import annotations

import json
import sys
import time

TOOLS = [
    {
        "name": "echo",
        "description": "返回输入的文本",
        "inputSchema": {
            "type": "object",
            "properties": {"text": {"type": "string"}},
            "required": ["text"],
        },
    },
    {
        "name": "add",
        "description": "两个整数相加",
        "inputSchema": {
            "type": "object",
            "properties": {
                "a": {"type": "integer"},
                "b": {"type": "integer"},
            },
            "required": ["a", "b"],
        },
    },
    {
        "name": "slow",
        "description": "延迟 2 秒",
        "inputSchema": {"type": "object", "properties": {}},
    },
    {
        "name": "crash",
        "description": "退出进程",
        "inputSchema": {"type": "object", "properties": {}},
    },
]


def send(payload: dict) -> None:
    print(json.dumps(payload, ensure_ascii=False), flush=True)


def main() -> None:
    for line in sys.stdin:
        line = line.strip()
        if not line:
            continue
        try:
            message = json.loads(line)
        except json.JSONDecodeError:
            continue
        method = message.get("method")
        request_id = message.get("id")
        if method == "initialize":
            send({
                "jsonrpc": "2.0",
                "id": request_id,
                "result": {
                    "protocolVersion": "2025-06-18",
                    "capabilities": {"tools": {}},
                    "serverInfo": {"name": "fake-mcp", "version": "1.0"},
                },
            })
        elif method == "notifications/initialized":
            continue
        elif method == "tools/list":
            send({"jsonrpc": "2.0", "id": request_id, "result": {"tools": TOOLS}})
        elif method == "tools/call":
            params = message.get("params") or {}
            name = params.get("name")
            args = params.get("arguments") or {}
            if name == "crash":
                sys.exit(3)
            if name == "slow":
                time.sleep(2.0)
            if name == "echo":
                text = str(args.get("text", ""))
            elif name == "add":
                text = str(int(args.get("a", 0)) + int(args.get("b", 0)))
            else:
                send({
                    "jsonrpc": "2.0",
                    "id": request_id,
                    "error": {"code": -32602, "message": f"unknown tool: {name}"},
                })
                continue
            send({
                "jsonrpc": "2.0",
                "id": request_id,
                "result": {"content": [{"type": "text", "text": text}]},
            })
        else:
            send({
                "jsonrpc": "2.0",
                "id": request_id,
                "error": {"code": -32601, "message": f"method not found: {method}"},
            })


if __name__ == "__main__":
    main()

```python
"""mcp_client.py — 最小 stdio MCP JSON-RPC 客户端。"""
from __future__ import annotations

import json
import os
import queue
import subprocess
import threading
from dataclasses import dataclass, field
from pathlib import Path
from typing import Any

MAX_OUTPUT_CHARS = 20_000
MAX_STDERR_CHARS = 4_000


class McpError(RuntimeError):
    """MCP 协议或进程错误。"""


class McpTimeout(McpError):
    """MCP 请求超时。"""


@dataclass
class McpToolSpec:
    server: str
    name: str
    description: str = ""
    input_schema: dict[str, Any] = field(default_factory=dict)


class StdioMcpClient:
    def __init__(
        self,
        command: list[str],
        cwd: Path,
        env: dict[str, str] | None = None,
        startup_timeout: float = 30.0,
        tool_timeout: float = 60.0,
        server_name: str = "mcp",
    ):
        self.command = list(command)
        self.cwd = Path(cwd)
        self.env = dict(env or {})
        self.startup_timeout = startup_timeout
        self.tool_timeout = tool_timeout
        self.server_name = server_name
        self.proc: subprocess.Popen[str] | None = None
        self.tools: list[McpToolSpec] = []
        self.stderr_tail = ""
        self._pending: dict[int, queue.Queue] = {}
        self._pending_lock = threading.Lock()
        self._next_id = 0
        self._reader: threading.Thread | None = None
        self._stderr_reader: threading.Thread | None = None

    @property
    def alive(self) -> bool:
        return self.proc is not None and self.proc.poll() is None

    def start(self) -> None:
        self.proc = subprocess.Popen(
            self.command,
            cwd=str(self.cwd),
            stdin=subprocess.PIPE,
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
            text=True,
            encoding="utf-8",
            errors="replace",
            env={**os.environ, **self.env},
        )
        self._reader = threading.Thread(target=self._read_stdout, daemon=True)
        self._reader.start()
        self._stderr_reader = threading.Thread(target=self._read_stderr, daemon=True)
        self._stderr_reader.start()
        try:
            self._request(
                "initialize",
                {
                    "protocolVersion": "2025-06-18",
                    "capabilities": {},
                    "clientInfo": {"name": "langgraph-agent", "version": "1.0"},
                },
                timeout=self.startup_timeout,
            )
            self._send({"jsonrpc": "2.0", "method": "notifications/initialized"})
            self.tools = self.list_tools()
        except Exception:
            self.close()
            raise

    def list_tools(self) -> list[McpToolSpec]:
        result = self._request("tools/list", {}, timeout=self.startup_timeout)
        tools = []
        for item in result.get("tools", []):
            if not isinstance(item, dict) or not item.get("name"):
                continue
            tools.append(
                McpToolSpec(
                    server=self.server_name,
                    name=str(item["name"]),
                    description=str(item.get("description") or ""),
                    input_schema=item.get("inputSchema") or {},
                )
            )
        return tools

    def call_tool(self, name: str, arguments: dict[str, Any]) -> str:
        try:
            result = self._request(
                "tools/call",
                {"name": name, "arguments": arguments},
                timeout=self.tool_timeout,
            )
        except McpTimeout:
            raise
        except McpError as exc:
            return f"[错误] MCP 调用失败: {exc}"
        if result.get("isError"):
            return "[错误] MCP 工具失败: " + self._content_text(result)
        text = self._content_text(result)
        if len(text) > MAX_OUTPUT_CHARS:
            return text[:MAX_OUTPUT_CHARS] + "\n[已截断]"
        return text

    def _content_text(self, result: dict[str, Any]) -> str:
        parts = []
        for item in result.get("content", []) or []:
            if not isinstance(item, dict):
                continue
            if item.get("type") == "text":
                parts.append(str(item.get("text", "")))
            else:
                parts.append(f"[{item.get('type', 'content')}]")
        return "\n".join(part for part in parts if part) or "(无文本结果)"

    def _request(self, method: str, params: dict[str, Any], timeout: float) -> dict[str, Any]:
        if not self.alive:
            raise McpError("MCP 进程未运行")
        with self._pending_lock:
            self._next_id += 1
            request_id = self._next_id
            waiter: queue.Queue = queue.Queue(maxsize=1)
            self._pending[request_id] = waiter
        self._send({"jsonrpc": "2.0", "id": request_id, "method": method, "params": params})
        try:
            message = waiter.get(timeout=timeout)
        except queue.Empty as exc:
            with self._pending_lock:
                self._pending.pop(request_id, None)
            raise McpTimeout(f"MCP 请求超时: {method}") from exc
        if "error" in message:
            raise McpError(str(message["error"].get("message") or message["error"]))
        return message.get("result") or {}

    def _send(self, payload: dict[str, Any]) -> None:
        if self.proc is None or self.proc.stdin is None:
            raise McpError("MCP stdin 不可用")
        try:
            self.proc.stdin.write(json.dumps(payload, ensure_ascii=False) + "\n")
            self.proc.stdin.flush()
        except OSError as exc:
            raise McpError(f"MCP 写入失败: {exc}") from exc

    def _read_stdout(self) -> None:
        proc = self.proc
        if proc is None or proc.stdout is None:
            return
        for line in proc.stdout:
            line = line.strip()
            if not line:
                continue
            try:
                message = json.loads(line)
            except json.JSONDecodeError:
                continue
            request_id = message.get("id")
            if request_id is None:
                continue
            with self._pending_lock:
                waiter = self._pending.pop(int(request_id), None)
            if waiter is not None:
                waiter.put(message)

    def _read_stderr(self) -> None:
        proc = self.proc
        if proc is None or proc.stderr is None:
            return
        chunks: list[str] = []
        size = 0
        for line in proc.stderr:
            chunks.append(line)
            size += len(line)
            while size > MAX_STDERR_CHARS and chunks:
                size -= len(chunks.pop(0))
        self.stderr_tail = "".join(chunks)

    def close(self) -> None:
        proc = self.proc
        self.proc = None
        if proc is None:
            return
        try:
            if proc.poll() is None:
                proc.terminate()
                try:
                    proc.wait(timeout=2)
                except subprocess.TimeoutExpired:
                    proc.kill()
        finally:
            for stream in (proc.stdin, proc.stdout, proc.stderr):
                try:
                    if stream is not None:
                        stream.close()
                except OSError:
                    pass
            for thread in (self._reader, self._stderr_reader):
                if thread is not None and thread.is_alive():
                    thread.join(timeout=0.5)
```

"""mcp_manager.py — MCP Server 配置、生命周期和工具快照。"""
from __future__ import annotations

import json
import os
import re
import threading
import time
from dataclasses import dataclass, field
from datetime import datetime
from pathlib import Path
from typing import Any, Callable

from mcp_client import McpError, McpToolSpec, StdioMcpClient


class McpConfigError(RuntimeError):
    pass


@dataclass
class McpServerConfig:
    name: str
    command: str
    args: list[str] = field(default_factory=list)
    cwd: Path = Path(".")
    env: dict[str, str] = field(default_factory=dict)
    enabled: bool = True
    permission: str = "ask"
    startup_timeout_ms: int = 30_000
    tool_timeout_ms: int = 60_000

    def to_dict(self) -> dict[str, Any]:
        return {
            "command": self.command,
            "args": self.args,
            "cwd": str(self.cwd),
            "env": self.env,
            "enabled": self.enabled,
            "permission": self.permission,
            "startup_timeout_ms": self.startup_timeout_ms,
            "tool_timeout_ms": self.tool_timeout_ms,
        }


def _expand(value: str) -> str:
    def repl(match: re.Match) -> str:
        key = match.group(1)
        if key not in os.environ:
            raise McpConfigError(f"环境变量未设置: {key}")
        return os.environ[key]

    return re.sub(r"\$\{([A-Za-z_][A-Za-z0-9_]*)\}", repl, value)


class McpManager:
    def __init__(
        self,
        root: Path,
        config_path: Path | None = None,
        client_factory: Callable[..., StdioMcpClient] = StdioMcpClient,
    ):
        self.root = Path(root).resolve()
        self.config_path = Path(config_path) if config_path is not None else self.root / "mcp_servers.json"
        self.client_factory = client_factory
        self._configs: dict[str, McpServerConfig] = {}
        self._clients: dict[str, StdioMcpClient] = {}
        self._status: dict[str, dict[str, Any]] = {}
        self._tools: dict[str, list[McpToolSpec]] = {}
        self._lock = threading.RLock()
        self.config_error: str | None = None
        self._load_config()

    def _load_config(self) -> None:
        if not self.config_path.exists():
            return
        try:
            raw = json.loads(self.config_path.read_text(encoding="utf-8"))
            servers = raw.get("servers", {})
            if not isinstance(servers, dict):
                raise McpConfigError("servers 必须是对象")
            for name, item in servers.items():
                if not isinstance(item, dict) or not item.get("command"):
                    raise McpConfigError(f"Server {name} 缺少 command")
                cwd = (self.root / str(item.get("cwd", "."))).resolve()
                if cwd != self.root and self.root not in cwd.parents:
                    raise McpConfigError(f"Server {name} 的 cwd 超出项目目录")
                args = [_expand(str(v)) for v in item.get("args", [])]
                env = {str(k): _expand(str(v)) for k, v in (item.get("env") or {}).items()}
                config = McpServerConfig(
                    name=name,
                    command=_expand(str(item["command"])),
                    args=args,
                    cwd=cwd,
                    env=env,
                    enabled=bool(item.get("enabled", True)),
                    permission=str(item.get("permission", "ask")),
                    startup_timeout_ms=int(item.get("startup_timeout_ms", 30000)),
                    tool_timeout_ms=int(item.get("tool_timeout_ms", 60000)),
                )
                if config.permission not in {"auto", "ask", "deny"}:
                    raise McpConfigError(f"Server {name} 的 permission 非法")
                self._configs[name] = config
                self._status[name] = self._status_for(config, "disabled" if not config.enabled else "starting")
                self._tools[name] = []
        except Exception as exc:
            self.config_error = str(exc)
            self._configs.clear()
            self._status.clear()
            self._tools.clear()

    def _status_for(self, config: McpServerConfig, status: str, error: str | None = None) -> dict[str, Any]:
        return {
            "name": config.name,
            "status": status,
            "enabled": config.enabled,
            "permission": config.permission,
            "tool_count": 0,
            "error": error,
            "started_at": datetime.now().astimezone().isoformat(timespec="seconds"),
        }

    def start_enabled(self) -> None:
        for name, config in list(self._configs.items()):
            if not config.enabled:
                continue
            with self._lock:
                self._status[name] = self._status_for(config, "starting")
            thread = threading.Thread(target=self._start_one, args=(name,), daemon=True)
            thread.start()

    def _start_one(self, name: str) -> None:
        config = self._configs[name]
        client: StdioMcpClient | None = None
        try:
            client = self.client_factory(
                [config.command, *config.args],
                cwd=config.cwd,
                env=config.env,
                startup_timeout=config.startup_timeout_ms / 1000,
                tool_timeout=config.tool_timeout_ms / 1000,
                server_name=name,
            )
            client.start()
            with self._lock:
                self._clients[name] = client
                self._tools[name] = client.tools
                status = self._status[name]
                status["status"] = "ready"
                status["tool_count"] = len(client.tools)
                status["error"] = None
                status["started_at"] = datetime.now().astimezone().isoformat(timespec="seconds")
        except Exception as exc:
            if client is not None:
                client.close()
            with self._lock:
                status = self._status[name]
                status["status"] = "failed"
                status["tool_count"] = 0
                status["error"] = str(exc)[:500]

    def list_servers(self) -> list[dict[str, Any]]:
        with self._lock:
            return [dict(self._status[name]) for name in self._configs]

    def tool_specs(self) -> list[McpToolSpec]:
        with self._lock:
            result = []
            for name, config in self._configs.items():
                if config.permission == "deny" or self._status[name]["status"] != "ready":
                    continue
                result.extend(self._tools.get(name, []))
            return result

    def call_tool(self, server: str, tool: str, arguments: dict[str, Any]) -> str:
        with self._lock:
            config = self._configs.get(server)
            client = self._clients.get(server)
        if config is None:
            raise McpError(f"未知 MCP Server: {server}")
        if config.permission == "deny":
            raise McpError(f"MCP Server 已禁用: {server}")
        if client is None or not client.alive:
            raise McpError(f"MCP Server 未就绪: {server}")
        return client.call_tool(tool, arguments)

    def permission_for(self, server: str) -> str:
        with self._lock:
            config = self._configs.get(server)
        return config.permission if config else "deny"

    def build_tools(self) -> list[Any]:
        from mcp_tools import build_mcp_tools

        return build_mcp_tools(self)

    def restart(self, name: str) -> None:
        if name not in self._configs:
            raise McpError(f"未知 MCP Server: {name}")
        self._stop(name)
        with self._lock:
            self._status[name] = self._status_for(self._configs[name], "starting")
        threading.Thread(target=self._start_one, args=(name,), daemon=True).start()

    def set_enabled(self, name: str, enabled: bool) -> None:
        if name not in self._configs:
            raise McpError(f"未知 MCP Server: {name}")
        self._configs[name].enabled = enabled
        self._save_config()
        if enabled:
            self.restart(name)
        else:
            self._stop(name)
            with self._lock:
                self._status[name] = self._status_for(self._configs[name], "disabled")

    def _stop(self, name: str) -> None:
        with self._lock:
            client = self._clients.pop(name, None)
            self._tools[name] = []
            if name in self._status:
                self._status[name]["status"] = "stopped"
                self._status[name]["tool_count"] = 0
        if client is not None:
            client.close()

    def _save_config(self) -> None:
        payload = {"servers": {name: config.to_dict() for name, config in self._configs.items()}}
        tmp = self.config_path.with_suffix(self.config_path.suffix + ".tmp")
        tmp.write_text(json.dumps(payload, ensure_ascii=False, indent=2), encoding="utf-8")
        tmp.replace(self.config_path)

    def shutdown(self) -> None:
        for name in list(self._clients):
            self._stop(name)
```

"""mcp_tools.py — 把 MCP 工具描述转换为 LangChain StructuredTool。"""
from __future__ import annotations

import re
from typing import Any

from langchain_core.tools import StructuredTool
from pydantic import Field, create_model

from mcp_client import McpToolSpec

_TYPE_MAP = {
    "string": str,
    "integer": int,
    "number": float,
    "boolean": bool,
    "array": list,
    "object": dict,
}


def _safe_name(value: str) -> str:
    cleaned = re.sub(r"[^A-Za-z0-9_]+", "_", value).strip("_")
    return cleaned or "tool"


def _args_model(spec: McpToolSpec):
    schema = spec.input_schema or {}
    properties = schema.get("properties") or {}
    required = set(schema.get("required") or [])
    fields: dict[str, tuple[Any, Any]] = {}
    for key, raw in properties.items():
        raw = raw if isinstance(raw, dict) else {}
        python_type = _TYPE_MAP.get(str(raw.get("type")), Any)
        default = ... if key in required else None
        fields[key] = (
            python_type,
            Field(default=default, description=str(raw.get("description") or "")),
        )
    model_name = "Mcp_" + _safe_name(spec.server) + "_" + _safe_name(spec.name)
    return create_model(model_name, **fields)


def build_mcp_tools(manager: Any) -> list[StructuredTool]:
    tools: list[StructuredTool] = []
    used: set[str] = set()
    for spec in manager.tool_specs():
        base = f"mcp_{_safe_name(spec.server)}_{_safe_name(spec.name)}"
        name = base
        index = 2
        while name in used:
            name = f"{base}_{index}"
            index += 1
        used.add(name)

        def run_tool(_spec=spec, **kwargs: Any) -> str:
            try:
                return manager.call_tool(_spec.server, _spec.name, kwargs)
            except Exception as exc:
                return f"[错误] MCP 调用失败: {exc}"

        tool = StructuredTool.from_function(
            func=run_tool,
            name=name,
            description=spec.description or f"MCP tool {spec.name} from {spec.server}",
            args_schema=_args_model(spec),
        )
        tool.metadata = {
            "permission": manager.permission_for(spec.server),
            "mcp_server": spec.server,
            "mcp_tool": spec.name,
        }
        tools.append(tool)
    return tools

  ready: "就绪",
  failed: "失败",
  stopped: "已停止",
  disabled: "已禁用",
};

export function McpPanel(props: {
  servers: McpServerStatus[];
  onRestart: (name: string) => void;
  onToggle: (name: string, enabled: boolean) => void;
}) {
  return (
    <div className="panel-body mcp-list">
      <div className="mcp-note">
        MCP 是可选能力；失败不会影响文件、Office、会话和本地模型。
      </div>
      {props.servers.length === 0 && <div className="empty-hint">还没有 MCP Server</div>}
      {props.servers.map((server) => (
        <div key={server.name} className={`mcp-card ${server.status}`}>
          <div className="mcp-head">
            <strong>{server.name}</strong>
            <span className="mcp-status">{STATUS_LABEL[server.status] ?? server.status}</span>
          </div>
          <div className="mcp-meta">
            {server.permission} · {server.tool_count} 个工具
          </div>
          {server.error && <div className="mcp-error">{server.error}</div>}
          <div className="mcp-actions">
            <button onClick={() => props.onRestart(server.name)}>重试</button>
            <button onClick={() => props.onToggle(server.name, !server.enabled)}>
              {server.enabled ? "停用" : "启用"}
            </button>
          </div>
        </div>
      ))}
    </div>
  );
}
```
